This loop for parsing the date was generated by AI:

```python
for feature in date_features:
    tokens = df_clean[feature].astype("string").str.strip()

    parsed = pd.Series(
        pd.NaT, index=df_clean.index, dtype="datetime64[ns]"
    )

    iso_mask = tokens.str.fullmatch(r"\d{4}-\d{2}-\d{2}", na=False)
    slash_mask = tokens.str.fullmatch(r"\d{2}/\d{2}/\d{4}", na=False)

    parsed.loc[iso_mask] = pd.to_datetime(
        tokens.loc[iso_mask], format="%Y-%m-%d", errors="coerce"
    )
    parsed.loc[slash_mask] = pd.to_datetime(
        tokens.loc[slash_mask], format="%d/%m/%Y", errors="coerce"
    )

    invalid = tokens.notna() & parsed.isna()

    if invalid.any():
        raise ValueError(
            f"Invalid dates in {feature}: "
            f"{tokens.loc[invalid].unique().tolist()}"
        )

    df_clean[feature] = parsed
```

```python
def missingness_associations(frame, targets, predictors):
    rows = []
    for target in targets:
        for predictor in predictors:
            if target == predictor: continue
            valid = frame[predictor].notna()
            table = pd.crosstab(frame.loc[valid, target].isna(), frame.loc[valid, predictor])
            if min(table.shape) < 2: continue
            chi2, _, _, expected = chi2_contingency(table, correction=False)
            n = table.to_numpy().sum(); r, k = table.shape
            if n <= 1: continue
            phi2 = max(0, chi2 / n - ((k - 1) * (r - 1)) / (n - 1))
            denominator = min(r - 1 - ((r - 1) ** 2) / (n - 1), k - 1 - ((k - 1) ** 2) / (n - 1))
            rows.append({"missing_feature": target, "predictor": predictor, "cramers_v": np.sqrt(phi2 / denominator) if denominator > 0 else np.nan, "n": int(n), "expected_cells_ge_5_pct": float((expected >= 5).mean() * 100)})

    return pd.DataFrame(rows, columns=["missing_feature", "predictor", "cramers_v", "n", "expected_cells_ge_5_pct"])
```
